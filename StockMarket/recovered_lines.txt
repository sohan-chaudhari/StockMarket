1186:   function toTimeNum(v) {
1187:     if (v == null) return 0;
1188:     if (typeof v === 'number') return v;
1189:     if (typeof v === 'string') {
1190:       var n = Number(v);
1191:       if (!isNaN(n)) return n;
1192:       // Parse date-only strings (YYYY-MM-DD) as UTC to avoid local timezone offset
1193:       if (/^\d{4}-\d{2}-\d{2}$/.test(v)) {
1194:         var ms = new Date(v + 'T00:00:00Z').getTime();
1195:         return isNaN(ms) ? 0 : Math.floor(ms / 1000);
1196:       }
1197:       var ms = new Date(v).getTime();
1198:       return isNaN(ms) ? 0 : Math.floor(ms / 1000);
1199:     }
